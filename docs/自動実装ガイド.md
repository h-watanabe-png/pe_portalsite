                    'columnCount': 10
                }
            }
        }
    }

    spreadsheetId=spreadsheet_id,

    spreadsheetId=spreadsheet_id,

    valueInputOption='RAW',
    body=body

    spreadsheetId=spreadsheet_id,

    body=body
).execute()

- `json` - JSONファイルの処理

- `gitpython` - Git操作（オプション、subprocessでも可）

### Git操作（Pythonスクリプトで自動化）

**Pythonスクリプトで実行**:
```python
import subprocess
import os

# 1. Gitリポジトリを初期化（既に存在する場合はスキップ）
if not os.path.exists('.git'):
    subprocess.run(['git', 'init'], check=True)

# 2. リモートリポジトリを設定
subprocess.run(['git', 'remote', 'add', 'origin', 'https://github.com/h-watanabe-png/pe_portalsite.git'], check=False)

# 3. 別ブランチ（ai-develop）を作成（既存プロジェクトのmainブランチとは分離）
subprocess.run(['git', 'checkout', '-b', 'ai-develop'], check=False)

# 3. .gitignoreを作成
gitignore_content = """node_modules/
.clasp.json
*.log
.DS_Store
"""
with open('.gitignore', 'w', encoding='utf-8') as f:
    f.write(gitignore_content)

# 4. 変更をステージング
subprocess.run(['git', 'add', '.'], check=True)

# 5. コミット
subprocess.run(['git', 'commit', '-m', '[自動] Phase 1: 基盤構築完了'], check=True)

# 6. リモートにプッシュ（必要に応じて、ai-developブランチに）
subprocess.run(['git', 'push', 'origin', 'ai-develop'], check=True)
```

**注意事項**:
- Gitがインストールされている必要がある
- 認証が完了している必要がある（SSH鍵またはHTTPS認証）
- エラーハンドリングを適切に実装する（リモートが既に存在する場合など）
