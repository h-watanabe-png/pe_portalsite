## 🎯 認証・認可の設計方針

### 方針1: Google OAuth認証 + ホワイトリスト方式（推奨）

**概要**: Googleアカウントでのログインを必須とし、ドメインチェックとホワイトリストでアクセス制御を行う。

#### メリット
- ✅ Googleアカウントのセキュリティを活用できる
- ✅ パスワード管理が不要（Googleが管理）
- ✅ ユーザー体験が良い（Googleログインのみ）
- ✅ GASの`Session.getActiveUser()`を活用できる

#### デメリット
- ⚠️ 外部アカウントもGoogleアカウントが必要
- ⚠️ ホワイトリストの管理が必要

#### 実装方法

```javascript
/**
 * 認証・認可チェック関数
 * @return {Object} {authorized: boolean, userEmail: string, reason: string}
 */
function checkAuthorization() {
  try {
    const user = Session.getActiveUser();
    const userEmail = user.getEmail();
    
    if (!userEmail) {
      return {
        authorized: false,
        userEmail: null,
        reason: 'ユーザーがログインしていません'
      };
    }
    
    // 会社ドメインのチェック
    const companyDomain = '@tomonokai-corp.com';
    if (userEmail.endsWith(companyDomain)) {
      return {
        authorized: true,
        userEmail: userEmail,
        reason: '会社ドメインのユーザー'
      };
    }
    
    // ホワイトリストのチェック
    const whitelist = getWhitelistEmails();
    if (whitelist.includes(userEmail)) {
      return {
        authorized: true,
        userEmail: userEmail,
        reason: 'ホワイトリストに登録されたユーザー'
      };
    }
    
    return {
      authorized: false,
      userEmail: userEmail,
      reason: 'アクセス権限がありません'
    };
  } catch (e) {
    Logger.log('認証チェックエラー: ' + e.toString());
    return {
      authorized: false,
      userEmail: null,
      reason: '認証チェック中にエラーが発生しました'
    };
  }
}

/**
 * ホワイトリストを取得（スプレッドシートから読み込み）
 * @return {Array<string>} 許可されたメールアドレスの配列
 */
function getWhitelistEmails() {
  const spreadsheetId = '1mivDNOXpZsE7oW7gF10Rq3NvnYjxjfiVQWp4LHIorL0';
  const sheetName = 'システム設定';
  
  try {
    const sheet = SpreadsheetApp.openById(spreadsheetId).getSheetByName(sheetName);
    const data = sheet.getDataRange().getValues();
    
    // ホワイトリストの設定項目を探す（例: 「外部ユーザーホワイトリスト」という設定項目）
    const whitelistRow = data.find(row => row[0] === '外部ユーザーホワイトリスト');
    if (whitelistRow && whitelistRow[1]) {
      // カンマ区切りでメールアドレスを分割
      return whitelistRow[1].split(',').map(email => email.trim()).filter(email => email);
    }
    
    return [];
  } catch (e) {
    Logger.log('ホワイトリスト取得エラー: ' + e.toString());
    return [];
  }
}

/**
 * Web Appのエントリーポイント（doGet）
 */
function doGet(e) {
  // 認証チェック
  const authResult = checkAuthorization();
  
  if (!authResult.authorized) {
    // アクセス拒否ページを返す
    return HtmlService.createHtmlOutput(`
      <!DOCTYPE html>
      <html>
      <head>
        <meta charset="UTF-8">
        <title>アクセス拒否</title>
      </head>
      <body>
        <h1>アクセス権限がありません</h1>
        <p>${authResult.reason}</p>
        <p>お問い合わせ: h-watanabe@tomonokai-corp.com</p>
      </body>
      </html>
    `);
  }
  
  // 認証成功時は通常のページを返す
  return HtmlService.createTemplateFromFile('index')
    .evaluate()
    .setTitle('PEポータルサイト')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}
```

### 方針2: カスタム認証（パスワード方式）

**概要**: Googleアカウントに依存せず、独自のパスワード認証を実装する。

#### メリット
- ✅ Googleアカウントが不要
- ✅ 完全に独自の認証システム
- ✅ 柔軟な認証ロジックを実装可能

#### デメリット
- ⚠️ パスワード管理が必要（ハッシュ化、ソルトなど）
- ⚠️ セキュリティリスクが高い（実装ミスで脆弱性が生じる可能性）
- ⚠️ ユーザー体験が悪い（パスワード入力が必要）
- ⚠️ GASの制約（6分制限、同時実行数制限）の影響を受けやすい

#### 実装方法（簡易版）

```javascript
/**
 * カスタム認証（パスワード方式）
 * 注意: 本番環境ではより強固な実装が必要
 */
function doGet(e) {
  // セッション確認
  const session = Session.getActive();
  const sessionToken = session.get('auth_token');
  
  if (!sessionToken) {
    // ログインページを表示
    return HtmlService.createTemplateFromFile('login')
      .evaluate()
      .setTitle('ログイン - PEポータルサイト');
  }
  
  // 認証済みの場合は通常のページを返す
  return HtmlService.createTemplateFromFile('index')
    .evaluate()
    .setTitle('PEポータルサイト');
}

/**
 * ログイン処理（doPost）
 */
function doPost(e) {
  const email = e.parameter.email;
  const password = e.parameter.password;
  
  // パスワード検証（簡易版）
  const authorizedUsers = getAuthorizedUsers();
  const user = authorizedUsers.find(u => u.email === email && u.password === password);
  
  if (user) {
    // セッションにトークンを保存
    const session = Session.getActive();
    session.put('auth_token', generateToken(email));
    
    return HtmlService.createHtmlOutput(`
      <script>
        window.top.location.href = '${ScriptApp.getService().getUrl()}';
      </script>
    `);
  } else {
    return HtmlService.createHtmlOutput(`
      <!DOCTYPE html>
      <html>
      <head>
        <meta charset="UTF-8">
        <title>ログイン失敗</title>
      </head>
      <body>
        <h1>ログインに失敗しました</h1>
        <p>メールアドレスまたはパスワードが正しくありません。</p>
        <a href="${ScriptApp.getService().getUrl()}">戻る</a>
      </body>
      </html>
    `);
  }
}

/**
 * 認証済みユーザーリストを取得（スプレッドシートから）
 */
function getAuthorizedUsers() {
  // スプレッドシートから認証情報を読み込む
  // 注意: パスワードはハッシュ化して保存する必要がある
  const spreadsheetId = '1mivDNOXpZsE7oW7gF10Rq3NvnYjxjfiVQWp4LHIorL0';
  const sheetName = '認証ユーザー管理';
  
  const sheet = SpreadsheetApp.openById(spreadsheetId).getSheetByName(sheetName);
  const data = sheet.getDataRange().getValues();
  
  // ヘッダー行をスキップ
  return data.slice(1).map(row => ({
    email: row[0],
    password: row[1], // 実際にはハッシュ化されたパスワード
    role: row[2]
  }));
}
```

**注意**: 本番環境では、パスワードのハッシュ化（bcrypt等）、ソルト、セッション管理の強化など、より強固な実装が必要です。

### 方針3: ハイブリッド方式（推奨・実用的）

**概要**: Google OAuth認証を基本とし、外部アカウントには一時的なアクセストークンを発行する。

#### メリット
- ✅ 会社ドメインはGoogle OAuthで簡単に認証
- ✅ 外部アカウントはトークンベースで柔軟に対応
- ✅ セキュリティと利便性のバランスが良い

#### 実装方法

```javascript
/**
 * ハイブリッド認証チェック
 */
function checkAuthorizationHybrid() {
  try {
    const user = Session.getActiveUser();
    const userEmail = user.getEmail();
    
    if (!userEmail) {
      // Googleアカウントでログインしていない場合、トークン認証を試行
      return checkTokenAuth();
    }
    
    // 会社ドメインのチェック
    const companyDomain = '@tomonokai-corp.com';
    if (userEmail.endsWith(companyDomain)) {
      return {
        authorized: true,
        userEmail: userEmail,
        authMethod: 'google_oauth',
        reason: '会社ドメインのユーザー'
      };
    }
    
    // ホワイトリストのチェック
    const whitelist = getWhitelistEmails();
    if (whitelist.includes(userEmail)) {
      return {
        authorized: true,
        userEmail: userEmail,
        authMethod: 'google_oauth',
        reason: 'ホワイトリストに登録されたユーザー'
      };
    }
    
    // Googleアカウントでログインしているが、アクセス権限がない場合
    return {
      authorized: false,
      userEmail: userEmail,
      authMethod: 'google_oauth',
      reason: 'アクセス権限がありません'
    };
  } catch (e) {
    Logger.log('認証チェックエラー: ' + e.toString());
    return checkTokenAuth();
  }
}

/**
 * トークン認証（URLパラメータから）
 */
function checkTokenAuth() {
  const e = ScriptApp.getService().getUrl();
  const token = e.parameter.token;
  
  if (!token) {
    return {
      authorized: false,
      userEmail: null,
      authMethod: 'token',
      reason: 'トークンが提供されていません'
    };
  }
  
  // トークンの検証（スプレッドシートから有効なトークンを確認）
  const validToken = validateToken(token);
  if (validToken) {
    return {
      authorized: true,
      userEmail: validToken.email,
      authMethod: 'token',
      reason: '有効なトークンで認証されました'
    };
  }
  
  return {
    authorized: false,
    userEmail: null,
    authMethod: 'token',
    reason: '無効または期限切れのトークンです'
  };
}

/**
 * トークンの検証
 */
function validateToken(token) {
  const spreadsheetId = '1mivDNOXpZsE7oW7gF10Rq3NvnYjxjfiVQWp4LHIorL0';
  const sheetName = '認証トークン管理';
  
  try {
    const sheet = SpreadsheetApp.openById(spreadsheetId).getSheetByName(sheetName);
    const data = sheet.getDataRange().getValues();
    
    // ヘッダー行をスキップ
    for (let i = 1; i < data.length; i++) {
      const row = data[i];
      const storedToken = row[0];
      const email = row[1];
      const expiresAt = new Date(row[2]);
      const isActive = row[3] === true || row[3] === 'TRUE';
      
      if (storedToken === token && isActive && expiresAt > new Date()) {
        return { email: email, token: token };
      }
    }
    
    return null;
  } catch (e) {
    Logger.log('トークン検証エラー: ' + e.toString());
    return null;
  }
}

/**
 * トークンを生成（管理者用関数）
 */
function generateAccessToken(email, daysValid = 30) {
  const token = Utilities.getUuid();
  const expiresAt = new Date();
  expiresAt.setDate(expiresAt.getDate() + daysValid);
  
  const spreadsheetId = '1mivDNOXpZsE7oW7gF10Rq3NvnYjxjfiVQWp4LHIorL0';
  const sheetName = '認証トークン管理';
  
  const sheet = SpreadsheetApp.openById(spreadsheetId).getSheetByName(sheetName);
  sheet.appendRow([token, email, expiresAt, true]);
  
  Logger.log(`トークンを生成しました: ${token} (有効期限: ${expiresAt})`);
  return token;
}
```

#### 使用方法

1. **会社ドメインのユーザー**: Googleアカウントでログインするだけでアクセス可能
2. **外部アカウント**: 管理者がトークンを生成し、URLに含めてアクセス
   - 例: `https://script.google.com/macros/s/.../exec?token=xxxx-xxxx-xxxx`

## 🔒 セキュリティ考慮事項

### 1. ホワイトリストの管理

- **推奨**: スプレッドシートの「システム設定」シートで管理
- **アクセス制御**: システム設定シートは管理者のみ編集可能にする
- **監査ログ**: ホワイトリストの変更を記録する

### 2. トークンの管理

- **有効期限**: デフォルト30日、必要に応じて短縮
- **無効化**: トークンが漏洩した場合、スプレッドシートから削除または無効化
- **定期クリーンアップ**: 期限切れトークンを自動削除するトリガーを設定

### 3. ログ記録

- **アクセスログ**: 認証成功・失敗を記録
- **異常検知**: 短時間に多数の認証失敗があった場合、アラートを送信

## 🎯 推奨実装

### 推奨: 方針1（Google OAuth + ホワイトリスト方式）

**理由**:
1. **セキュリティ**: Googleアカウントのセキュリティを活用できる
2. **利便性**: ユーザーはGoogleログインのみでアクセス可能
3. **実装の簡単さ**: GASの標準機能を活用できる
4. **GASの制約への対応**: 6分制限や同時実行数制限の影響を受けにくい

### 実装手順

1. **スプレッドシートにホワイトリストを追加**
   - 「システム設定」シートに「外部ユーザーホワイトリスト」を追加

2. **認証チェック関数を実装**
   - `checkAuthorization()`関数を実装

3. **Web Appのデプロイ設定**
   - 「Webアプリケーションにアクセスしているユーザー（User accessing the web app）」でデプロイ
   - これにより、各ユーザーが自分のGoogleアカウントでログインする

4. **テスト**
   - 会社ドメインのユーザーでアクセステスト
   - ホワイトリストに追加した外部アカウントでアクセステスト
